# Photon material library - Phase 1 prototype set.
#
# Units: SI. Temperatures in kelvin.
# PROTOTYPE DATA: values are typical literature values used to get a running
# prototype. Each carries its source; status for every value is tracked in
# docs/DATA_REGISTER.md. Values marked "verify" must be checked against the
# primary source before Phase 4 replaces the prototype burning model.
#
# Burning keys (prototype model, see docs/PROTOTYPE_PHYSICS.md):
#   pyrolysis_temp        surface temperature where off-gassing begins
#   heat_of_gasification  energy to turn 1 kg of solid into off-gas
#   heat_of_combustion    effective heat released per kg of off-gas burned
#   soot_yield            kg soot per kg burned (well-ventilated)
#   aerosol_yield         kg light-colored pyrolysis aerosol per kg off-gassed
#   char_yield            kg char / inert residue left per kg of solid
#   surface_depth         thickness of the heated surface node (prototype lumping)
#   fail_temp / fail_time non-combustible layer failure (fall-off, crack-out, melt)
#   offgas <gas> <mass fraction>  what the material gives off when heated (gases in
#                         data/gases.txt). Fractions sum to 1. PROTOTYPE product
#                         splits: Phase 4 data job replaces them with measured
#                         pyrolysis product data per material.

material gypsum_board
  name "Gypsum board (regular, Type X shares values)"
  density 640            # 1/2 in board ~1.6-1.8 lb/ft2 -> ~ 640-700 kg/m3
  conductivity 0.17
  specific_heat 1090
  emissivity 0.9
  combustible no
  fail_temp 923          # prototype: calcined through and falling off at ~650 C
  fail_time 60
  color 0.88 0.87 0.84
  source "density/k/cp: SFPE Handbook 5th ed., thermal property appendix (verify); Gypsum Association GA-235 for board mass"
  source "fail_temp/fail_time: PROTOTYPE estimate, replaced by dehydration + fall-off model in Phase 4"
  offgas H2O 1.0  # chemically bound water driven off as steam (~21% of board mass), dehydration only
  # structural (attached boards): ASTM C1396 nail pull resistance, 1/2 in regular board: 77 lbf = 343 N per
  # fastener. Ceiling attachment about 5.5 fasteners per m2 (screws 12 in o.c. on framing 24 in o.c., GA-216).
  # Heat reduces it as the board dehydrates (integrity from the fire model).
  fastener_pull 343
  fastener_density 5.5
  source "Structural: ASTM C1396 Table (nail pull resistance) (verify); fastening per Gypsum Association GA-216 (verify)"
end
